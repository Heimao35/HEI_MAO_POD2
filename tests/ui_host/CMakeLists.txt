cmake_minimum_required(VERSION 3.16)
project(player_ui_regressions C)
set(CMAKE_C_STANDARD 11)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LVGL "${REPO}/managed_components/lvgl__lvgl")
if(NOT EXISTS "${LVGL}/src/lvgl.h")
  message(FATAL_ERROR "Run the ESP-IDF configure first to fetch the pinned LVGL dependency")
endif()
file(GLOB_RECURSE LVGL_C "${LVGL}/src/*.c")
add_library(lvgl_host STATIC ${LVGL_C})
target_include_directories(lvgl_host PUBLIC "${LVGL}" "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(lvgl_host PUBLIC LV_KCONFIG_IGNORE LV_CONF_INCLUDE_SIMPLE)
add_executable(player_ui_tests test_player.c)
target_include_directories(player_ui_tests PRIVATE stubs ../media_host/stubs "${REPO}/main")
target_link_libraries(player_ui_tests PRIVATE lvgl_host)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(lvgl_host PRIVATE -O1 -ffunction-sections -fdata-sections)
  target_compile_options(player_ui_tests PRIVATE -O1 -ffunction-sections -fdata-sections)
  target_link_options(player_ui_tests PRIVATE -Wl,--gc-sections -static-libgcc)
  if(NOT WIN32)
    target_link_libraries(player_ui_tests PRIVATE m)
  endif()
endif()
enable_testing()
add_test(NAME player_ui COMMAND player_ui_tests)
